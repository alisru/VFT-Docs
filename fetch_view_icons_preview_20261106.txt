      document.documentElement.style.setProperty('--icon-size', e.target.value);
    };
    themeSelect.onchange = e => {
      document.body.className = '';
      if (e.target.value === 'light') document.body.className = 'light-theme';
      if (e.target.value === 'classic') document.body.className = 'classic-blue';
    };
  </script>
